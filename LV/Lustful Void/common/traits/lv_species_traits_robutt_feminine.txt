# Maternal Programming
trait_lv_maternal_programming = {
	cost = {
		base = 3
		modifier = {
			add = -3
			has_country_flag = lv_cheat_mode
		}
	}
	allowed_archetypes = { ROBOT MACHINE }
	custom_tooltip = trait_lv_maternal_programming_effect
	
	opposites = {
		trait_lv_paternal_programming
	}
	
	# Synth and Machine only
	species_potential_add = {
		exists = from
		from = {
			OR = {
				has_technology = tech_lv_biological_mimicry_m
				has_technology = tech_synthetic_leaders
				has_country_flag = lv_cheat_mode
			}
			NOT = { has_policy_flag = ai_outlawed }
		}
	}
	
	modifier = {
		# Protective
		army_defense_health_mult = 0.1
		army_defense_morale_mult = 0.1
		army_defense_damage_mult = 0.1
		# Nurturing
		planet_jobs_food_produces_mult = 0.15
		pop_housing_usage_mult = -0.15
		pop_resettlement_cost_mult = -0.25
	}
	
	slave_cost = {
		energy = 150
	}
}
