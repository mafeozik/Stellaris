# Seductive 1
leader_trait_lv_seductive = {
	inline_script = {
		script = trait/icon
		CLASS = leader
		ICON = "GFX_leader_trait_lv_seductive"
		RARITY = veteran
		COUNCIL = yes
		TIER = 1
	}
	
	# 'Loved' by the people
	triggered_sector_modifier = {
		potential = { trait_is_gestalt_check = no }
		pop_happiness = 0.05
	}
	triggered_councilor_modifier = {
		potential = {
			owner = { is_gestalt = no }
		}
		faction_approval = 0.025
	}
	triggered_councilor_modifier = {
		potential = {
			owner = { is_gestalt = yes }
		}
		country_unity_produces_mult = 0.05
	}
	# Knows exactly what to say
	councilor_modifier = {
		country_edict_fund_add = 25
		country_trust_cap_add = 25
	}
	
	leader_potential_add = {
		has_skill > 1
		OR = {
			is_gestalt = no
			AND = {
				is_gestalt = yes
				is_councilor = yes
			}
		}
		OR = {
			owner = { is_machine_empire = no }
			species = { lv_is_a_smexy_robutt = yes }
		}
	}
	
	leader_class = { commander scientist official }
	selectable_weight = {
		weight = @shared_trait_weight
		inline_script = "paragon/council_weight_mult"
	}
}

# Pain Slut 1
leader_trait_lv_pain_slut = {
	inline_script = {
		script = trait/icon
		CLASS = leader
		ICON = "GFX_leader_trait_lv_pain_slut"
		RARITY = common
		COUNCIL = no
		TIER = 1
	}
	
	# Anything can be a toy if you're brave enough!
	triggered_self_modifier = {
		potential = {
			exists = owner
			OR = {
				AND = {
					owner = {
						is_synthetic_empire = no
					}
					is_hive_species = yes
				}
				is_immortal = no
			}
		}
		leader_lifespan_add = -20
	}
	self_modifier = {
		leader_trait_selection_options_add = 2
		species_leader_exp_gain = 0.2
	}
	
	leader_potential_add = {
		has_skill > 1
		OR = {
			is_gestalt = no
			AND = {
				is_gestalt = yes
				is_councilor = yes
			}
		}
		OR = {
			owner = { is_machine_empire = no }
			species = { lv_is_a_smexy_robutt = yes }
		}
	}
	
	leader_class = { commander scientist official }
	selectable_weight = {
		weight = @shared_trait_weight
		modifier = {
			species = {
				has_trait = trait_lv_masochistic
			}
			factor = @species_trait_weight_mult
		}
	}
}
