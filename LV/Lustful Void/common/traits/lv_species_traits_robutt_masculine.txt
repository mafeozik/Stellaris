# Paternal Programming
trait_lv_paternal_programming = {
	cost = {
		base = 3
		modifier = {
			add = -3
			has_country_flag = lv_cheat_mode
		}
	}
	allowed_archetypes = { ROBOT MACHINE }
	custom_tooltip = trait_lv_paternal_programming_effect
	
	opposites = {
		trait_lv_maternal_programming
	}
	
	# Synth and Machine only
	species_potential_add = {
		exists = from
		from = {
			OR = {
				has_technology = tech_lv_biological_mimicry_m
				has_technology = tech_synthetic_leaders
				has_country_flag = lv_cheat_mode
			}
			NOT = { has_policy_flag = ai_outlawed }
		}
	}
	
	modifier = {
		# Lead by example
		army_damage_mult = 0.2
		army_attack_morale_mult = 0.1
		# Hard work pays off
		planet_jobs_energy_produces_mult = 0.15
		planet_jobs_minerals_produces_mult = 0.25
	}
	
	slave_cost = {
		energy = 150
	}
}
