# Sexologist 1
leader_trait_scientist_lv_sexologist = {
	inline_script = {
		script = trait/icon
		CLASS = scientist
		ICON = "GFX_leader_trait_lv_sexologist"
		RARITY = common
		COUNCIL = yes
		TIER = 1
	}
	
	councilor_modifier = {
		# Tries to find a sexual use first
		all_technology_research_speed = 0.05
	}
	modifier = {
		# Very thorough
		ship_anomaly_generation_chance_mult = 0.15
	}
	
	leader_potential_add = {
		OR = {
			owner = { is_machine_empire = no }
			species = { lv_is_a_smexy_robutt = yes }
		}
	}
	
	leader_class = { scientist }
	selectable_weight = @class_trait_weight
}
