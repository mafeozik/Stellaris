		script = trait/icon
		CLASS = official
		ICON = "GFX_leader_trait_lv_family_planner"
		RARITY = veteran
		COUNCIL = yes
		TIER = 1
	}

		# Encourages experimentation
		planet_housing_add = 5
	}

		planet_jobs_unity_produces_mult = 0.05
	}

		planet_jobs_society_research_produces_mult = 0.05
	}

		}
		has_paragon_dlc = yes
	}
	veteran_class_locked_trait = yes

}
