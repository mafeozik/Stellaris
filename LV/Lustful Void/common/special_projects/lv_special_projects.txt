#Extract Rogue Sexy Scientist
special_project = {
	key = "LV_EXTRACT_ROGUE_SEXY_SCIENTIST"
	cost = 0
	days_to_research = 180
	tech_department = society_technology
	timelimit = 1080
	picture = GFX_evt_gas_giant
	
	event_scope = ship_event
	
	requirements = {
		shipclass_transport = 2
	}
	
	on_success = {
		ship_event = { id = lv_observation_events.10 }
	}
	
	on_fail = {
		owner = {
			country_event = { id = lv_observation_events.20 }
		}
	}
	
	abort_trigger = {
		exists = event_target:pre_ftl_planet
		event_target:pre_ftl_planet = {
			OR = {
				NOT = { exists = event_target:pre_ftl_country }
				AND = {
					exists = owner
					owner = {
						NOT = {
							is_same_value = event_target:pre_ftl_country
						}
					}
				}
			}
		}
	}
}
