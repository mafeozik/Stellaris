@r_lv_activation_cost = 3000
@r_lv_duration = 3600
@r_lv_duration_short = 1800
@r_lv_duration_veryshort = 900

# The Golden Rod #
r_lv_golden_rod = {
	activation_duration = @r_lv_duration_veryshort
	portrait = "GFX_relic_lv_golden_rod"
	sound = "relic_activation_generic"

	resources = {
		category = relics
		# Activation cost
		cost = {
			unity = @r_lv_activation_cost
		}
	}
	
	ai_weight = {
		weight = 100
	}
	
	triggered_country_modifier = {
		potential = {
			always = yes
		}
		country_society_tech_research_speed = 0.2
		pop_growth_speed = 0.2
	}

	score = 1500

	active_effect = {
		custom_tooltip = relic_lv_golden_rod_active
		custom_tooltip = relic_lv_duration_veryshort_cooldown
		hidden_effect = {
			add_modifier = {
				modifier = "relic_activation_cooldown"
				days = @r_lv_duration_veryshort
			}
			country_event = {
				id = lv_relic_events.11
			}
		}
	}

	# Possible check for activation
	possible = {
		custom_tooltip = {
			fail_text = "requires_relic_no_cooldown"
			NOT = { has_modifier = relic_activation_cooldown }
		}	
	}
}
