# Pleasure Habitat #

district_lv_shabitat_milk = {
	base_buildtime = @hab_time
	is_capped_by_modifier = no

	show_on_uncolonized = {
		uses_district_set = habitat
		has_planet_flag = lv_pleasure_habitat
	}

	potential = {
		exists = owner
		owner = { has_technology = tech_lv_lactation_drugs }
		uses_district_set = habitat
		has_planet_flag = lv_pleasure_habitat
	}
	
	resources = {
		category = planet_districts
		cost = {
			minerals = @hab_cost
		}
		upkeep = {
			energy = @hab_maintenance
			alloys = @low_hab_maintenance
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = yes }
		}
		modifier = {
			job_lv_milk_drone_add = 5
			planet_housing_add = 5
			mult = owner.value:hab_void_dweller_jobs
		}
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = no }
		}
		modifier = {
			job_lv_milk_cow_add = 5
			planet_housing_add = 5
			mult = owner.value:hab_void_dweller_jobs
		}
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = { is_hive_empire = yes }
		}
		text = job_lv_milk_drone_effect_desc
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = { is_hive_empire = no }
		}
		text = job_lv_milk_cow_effect_desc
	}
}
