# Dairy #

planet_lv_milk_cow = {
	icon = "job_lv_milk_cow"
	parent = planet_livestock
	generate_mult_modifiers = {
		produces
		upkeep
	}
	generate_add_modifiers = {
		produces
		upkeep
	}
	modifier_category = pop

	## workaround to accept add modifier
	triggered_produces_modifier = {
		key = planet_livestock
		modifier_types = {
			add
		}
		trigger = {
			always = yes
		}
	}
	## milk job trigger
	triggered_produces_modifier = {
		key = planet_lv_dairy_jobs
		modifier_types = {
			mult
			add
		}
		trigger = {
			OR = {
				is_pop_category = worker
				AND = {
					is_pop_category = slave # Yes, slave should count towards both worker and slave
					has_job_category = worker
				}
				is_pop_category = simple_drone
			}
		}
	}
}
