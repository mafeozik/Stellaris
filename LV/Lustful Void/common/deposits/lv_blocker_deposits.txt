d_lv_tentacles_blocker = {
	time = 270
	
    is_for_colonizable = yes
	category = deposit_cat_rare_blocker

	resources = {
		category = deposit_blockers
		cost = {
			food = 1000
		}
	}
	
	potential = {
		# Gigastructural Engineering compatibility
		giga_has_birch_world = no
	}
	
	planet_modifier = { 
		planet_max_districts_add = -1
	}
	
	use_weights_for_blocker_swap_types = no
	blocker_swap_types = {
		d_lv_tentacle_symbiosis
	}

	drop_weight = {
		weight = 5
		modifier = {
			factor = 0.5
			planet_size < 15
		}
		modifier = {
			factor = 0
			num_free_districts = {
				type = district_city
				value < 2
			}
		}
	}

	prerequisites = { "tech_lv_tentacle_symbiosis" }
}
