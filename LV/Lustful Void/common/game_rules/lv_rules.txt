lv_can_fill_dairy_worker_job = {
	custom_tooltip = WORKER_JOB_TRIGGER
	hidden_trigger = {
		exists = owner
		owner = { is_gestalt = no }
		is_being_purged = no
		is_being_assimilated = no
		OR = {
			NOT = { has_trait = trait_mechanical }
			has_trait = trait_lv_synth_breasts
			has_trait = trait_lv_sex_bots
			has_trait = trait_lv_sex_bots_o
			has_trait = trait_lv_synth_penis
			has_trait = trait_lv_lushtech_omni_seed_fabrication_modules
		}
		#OR = {		# To be added when cum studs are made
		#	NOR = {
		#		species = { species_gender = male }
		#		has_trait = trait_lv_all_male
		#	}
		#	has_trait = trait_lv_excess_estrogen
		#}
		if = {
			limit = {
				divinity_right_to_work_job_check_trigger_exempt = no
			}
			divinity_right_to_work_job_check_trigger_worker = yes
		}
	}
}
