# Regular #

# Comfort Zone
building_lv_comfort_zone = {
	base_buildtime = @b1_time
	
	category = amenity
	
	potential = {
		exists = owner
		owner = { is_gestalt = no }
		OR = {
			owner = {
				NOT = { is_ai = yes }
			}
			AND = {
				NOT = { has_available_jobs = lv_comfort_worker }
				NOT = { has_forbidden_jobs = lv_comfort_worker }
			}
		}
		NOT = { has_building = building_low_tech_capital }
	}
	
	destroy_trigger = {
		exists = owner
		owner = { is_gestalt = yes }
	}
	
	resources = {
        category = planet_buildings
        cost = {
			minerals = @b1_minerals
		}
        upkeep = {
			energy = @b1_upkeep
		}
	}
	
	upgrades = {
		building_lv_comfort_megaplex
	}
	
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		modifier = {
			job_lv_comfort_worker_add = @b1_jobs
		}
	}
	
	triggered_desc = {
        trigger = {
            always = yes
        }
        text = job_lv_comfort_worker_effect_desc
	}
}

# Comfort Megaplex
building_lv_comfort_megaplex = {
	base_buildtime = @b2_time
	can_build = no
	
	category = amenity
	
	potential = {
		has_building = building_lv_comfort_zone
	}
	
	allow = {
		has_upgraded_capital = yes
		OR = {
			owner = {
				NOT = { is_ai = yes }
			}
			AND = {
				NOT = { has_available_jobs = entertainer }
				NOT = { has_forbidden_jobs = entertainer }
			}
		}
	}
	
	destroy_trigger = {
		exists = owner
		owner = { is_gestalt = yes }
	}

	resources = {
        category = planet_buildings
        cost = {
			minerals = @b2_minerals
			exotic_gases = @b2_rare_cost
		}
        upkeep = {
			energy = @b2_upkeep
			exotic_gases = @b2_rare_upkeep
		}
	}
	
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		modifier = {
			job_lv_comfort_worker_add = @b2_jobs
			job_lv_comfort_worker_per_pop = 0.05
		}
	}
	
	triggered_desc = {
        trigger = {
            always = yes
        }
        text = job_lv_comfort_worker_effect_desc
	}
}
