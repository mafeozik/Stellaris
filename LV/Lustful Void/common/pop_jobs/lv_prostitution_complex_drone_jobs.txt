# Stimulator Drone

lv_stimulator_drone = {
	category = complex_drone
	condition_string = DRONE_JOB_TRIGGER
	building_icon = building_lv_breeder_nh
	
	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
	
	possible_precalc = can_fill_drone_job
	
	possible = { NOT = { has_trait = trait_nerve_stapled } }
	
	resources = {
        category = planet_lv_stimulator
		
		# Any
		produces = {
			energy = 12
		}
		produces = {
			trigger = {
				exists = owner
				owner = {
					is_robot_empire = yes
				}
			}
			energy = 4
		}
		produces = {
			trigger = {
				planet = {
					has_planet_flag = has_energy_requisitorium
				}
			}
			energy = @requisitorium_energy_reduction
		}
		produces = {
			trigger = {
				exists = owner
				owner = { has_policy_flag = lv_prostitution_supported }
			}
			unity = 2
		}
		produces = {
			trigger = {
				OR = {
					pop_has_trait = trait_lv_sexual_pheromones
					pop_has_trait = trait_lv_pheromone_generators
					pop_has_trait = trait_lv_sex_bots
					pop_has_trait = trait_lv_sex_bots_o
					pop_has_trait = trait_lv_aphrodisiac_milk
					pop_has_trait = trait_lv_aphrodisiac_cum
				}
			}
			energy = 1
		}
		produces = {
			trigger = {
				OR = {
					pop_has_trait = trait_lv_targeted_pheromones
					pop_has_trait = trait_lv_addictive_cum
					pop_has_trait = trait_lv_perfect_hybrid
					pop_has_trait = trait_lv_maternal_programming
					pop_has_trait = trait_lv_paternal_programming
				}
			}
			energy = 2
		}
	}
	
	overlord_resources = {
		category = planet_requisitioned_energy
		produces = {
			trigger = {
				planet = {
					has_planet_flag = has_energy_requisitorium
				}
			}
			energy = @requisitorium_energy_overlord
		}
	}
	
	weight = {
		weight = 10
		mult = value:job_weights_modifier|JOB|lv_stimulator_drone|RESOURCE|energy|
		# factors
		modifier = {
			factor = 1.2
			has_job = lv_stimulator_drone
		}
		
		# low income modifier
		modifier = {
			factor = 1.2
			exists = planet
			planet = { has_available_jobs = lv_stimulator }
			exists = owner
			owner = {
				is_ai = yes
				has_monthly_income = {
					resource = energy
					value < 10
				}
			}
		}
		
		# sex traits
		modifier = {
			add = 0.05
			lv_sex_weight_low = yes
		}
		modifier = {
			add = 0.1
			lv_sex_weight_med = yes
		}
		modifier = {
			add = 0.2
			lv_sex_weight_high = yes
		}
		modifier = {
			add = -0.2
			lv_sex_weight_neg = yes
		}
	}
}
