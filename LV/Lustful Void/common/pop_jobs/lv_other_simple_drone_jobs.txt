lv_nurture_drone = {			# Nurture Drone
	category = simple_drone
	condition_string = DRONE_JOB_TRIGGER
	building_icon = building_lv_nurture_megaplex
	
	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
	
	possible_precalc = can_fill_drone_job
	
	resources = {
        category = planet_lv_comfort_worker
		
		# Organic (or robutts with proper parts)
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_technology = tech_lv_lactation_drugs
				}
				is_lithoid = no
				OR = {
					pop_has_trait = trait_lv_multiple_breasts
					pop_has_trait = trait_lv_fuckable_nipples
					pop_has_trait = trait_lv_high_volume_lactation
					pop_has_trait = trait_lv_highly_nutritious_milk
					pop_has_trait = trait_lv_fizzy_milk
					pop_has_trait = trait_lv_titanic_evolution
				}
			}
            food = 1
		}
		
		# Lithoids
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_technology = tech_lv_lactation_drugs
				}
				is_lithoid = yes
				OR = {
					pop_has_trait = trait_lv_multiple_breasts
					pop_has_trait = trait_lv_fuckable_nipples
					pop_has_trait = trait_lv_high_volume_lactation
					pop_has_trait = trait_lv_highly_nutritious_milk
					pop_has_trait = trait_lv_fizzy_milk
					pop_has_trait = trait_lv_titanic_evolution
				}
			}
            minerals = 1
		}
		
		# Robutts
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_technology = tech_lv_lactation_drugs
				}
				OR = {
					pop_has_trait = trait_lv_synth_breasts
					pop_has_trait = trait_lv_sex_bots
					pop_has_trait = trait_lv_sex_bots_o
					pop_has_trait = trait_lv_synth_penis
					pop_has_trait = trait_lv_lushtech_omni_seed_fabrication_modules
				}
			}
            energy = 1
		}
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_valid_civic = civic_lv_machine_nurture_protocols
				}
			}
			unity = 1
		}
		
		# Any
		produces = {
			trigger = {
				OR = {
					pop_has_trait = trait_lv_targeted_pheromones
					pop_has_trait = trait_lv_addictive_cum
					pop_has_trait = trait_lv_perfect_hybrid
				}
			}
			unity = 1
		}
	}
	
	# Any
	planet_modifier = {
		planet_jobs_complex_drone_produces_mult = 0.01
	}
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_no_happiness_add = 4
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
	}
	triggered_planet_modifier = {
		potential = {
			owner = { has_active_tradition = tr_lv_nurturing_finish }
		}
		planet_amenities_no_happiness_add = 2
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	triggered_planet_modifier = {
		potential = {
			exists = planet.orbital_defence
			planet.orbital_defence = {
				exists = starbase
				starbase = {
					has_starbase_building = lv_ring_nurture_hub
				}
			}
		}
		planet_amenities_no_happiness_add = 1
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	triggered_planet_modifier = {
		potential = {
			planet = {
				is_ringworld = yes
				has_building = building_alpha_hub
			}
		}
		planet_amenities_no_happiness_add = 1
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	
	weight = {
		weight = @maintenance_drone_job
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		# factors
		modifier = {
			factor = 1.2
			has_job = lv_nurture_drone
		}
		modifier = {
			mult = value:job_weights_modifier|JOB|lv_nurture_drone|RESOURCE|unity|FACTOR|0.25|
			exists = owner
			owner = {
				has_valid_civic = civic_lv_machine_nurture_protocols
			}
		}
		modifier = {
			factor = 1.1
			exists = owner
			years_passed < 1
			owner = { has_valid_civic = civic_machine_assimilator }
			has_trait = trait_cybernetic
		}
		modifier = {
			factor = 1000 # <- hive mind amenities jobs are non specialist so they have base weight 1 instead of 1000 but we still want them
			NOT = { has_job = lv_nurture_drone }
			planet = {
				free_amenities < 3
			}
		}
		modifier = {
			factor = 1000 # <- hive mind amenities jobs are non specialist so they have base weight 1 instead of 1000 but we still want them
			has_job = lv_nurture_drone
			planet = {
				free_amenities < 8
			}
		}
		
		# sex traits
		modifier = {
			add = 0.05
			lv_sex_weight_low = yes
		}
		modifier = {
			add = 0.1
			lv_sex_weight_med = yes
		}

			add = 0.2
			lv_sex_weight_high = yes
		}
		modifier = {
			add = -0.2
			lv_sex_weight_neg = yes
		}
	}
}

	category = simple_drone
	condition_string = DRONE_JOB_TRIGGER

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_drone_job

	possible = {
		NOT = { pop_has_trait = trait_lv_equine_penis }
	}

	resources = {
		category = planet_lv_rod_temple_initiate
		
		# Organic
		upkeep = {
			trigger = {
				is_organic_species = yes
				is_lithoid = no
			}
			food = 4
		}
		
		# Lithoids
		upkeep = {
			trigger = {
				is_lithoid = yes
			}
			minerals = 4
		}
		
		# Any
		produces = {
			unity = 10
			society_research = 20
		}
	}

	# Any
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_no_happiness_add = 5
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
	}
	
	weight = {
		weight = @simple_drone_job_weight
		mult = value:job_weights_multiple_resources_modifier|JOB|lv_rod_temple_initiate_drone|RESOURCE1|unity|WEIGHT1|0.5|RESOURCE2|society_research|WEIGHT2|0.3|
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		# factors
		modifier = {
			factor = 0.5
			has_citizenship_rights = no  #willing apprentices first, please
			NOT = { has_trait = trait_mechanical }
		}

			factor = 30

		}
		modifier = {
			factor = 0
			has_trait = trait_lv_equine_penis
		}
	}
}
