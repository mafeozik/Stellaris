lv_comfort_worker = {			# Comfort Worker
	category = worker
	condition_string = WORKER_JOB_TRIGGER
	building_icon = building_lv_comfort_megaplex
	clothes_texture_index = 2
	
	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
	
	possible_precalc = can_fill_worker_job
	
	resources = {
        category = planet_lv_comfort_worker
		
		# Organic (or robutts with proper parts)
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_technology = tech_lv_lactation_drugs
				}
				is_lithoid = no
				OR = {
					pop_has_trait = trait_lv_multiple_breasts
					pop_has_trait = trait_lv_fuckable_nipples
					pop_has_trait = trait_lv_high_volume_lactation
					pop_has_trait = trait_lv_highly_nutritious_milk
					pop_has_trait = trait_lv_fizzy_milk
					pop_has_trait = trait_lv_titanic_evolution
				}
			}
            food = 1
		}
		
		# Lithoids
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_technology = tech_lv_lactation_drugs
				}
				is_lithoid = yes
				OR = {
					pop_has_trait = trait_lv_multiple_breasts
					pop_has_trait = trait_lv_fuckable_nipples
					pop_has_trait = trait_lv_high_volume_lactation
					pop_has_trait = trait_lv_highly_nutritious_milk
					pop_has_trait = trait_lv_fizzy_milk
					pop_has_trait = trait_lv_titanic_evolution
				}
			}
            minerals = 1
		}
		
		# Robutts
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_technology = tech_lv_lactation_drugs
				}
				OR = {
					pop_has_trait = trait_lv_synth_breasts
					pop_has_trait = trait_lv_sex_bots
					pop_has_trait = trait_lv_sex_bots_o
					pop_has_trait = trait_lv_synth_penis
					pop_has_trait = trait_lv_lushtech_omni_seed_fabrication_modules
				}
			}
            energy = 1
		}
		
		# Any
		produces = {
			trigger = {
				exists = owner
				owner = { has_valid_civic = civic_lv_sexculture }
			}
			unity = 1
		}
		produces = {
			trigger = {
				OR = {
					pop_has_trait = trait_lv_targeted_pheromones
					pop_has_trait = trait_lv_addictive_cum
					pop_has_trait = trait_lv_perfect_hybrid
				}
			}
			unity = 1
		}
	}
	
	# Any
	planet_modifier = {
		planet_jobs_specialist_produces_mult = 0.01
		planet_jobs_ruler_produces_mult = 0.01
	}
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_add = 4
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	triggered_planet_modifier = {
		potential = {
			planet = {
				exists = orbital_defence
				orbital_defence = {
					exists = starbase
					starbase = {
						has_starbase_building = lv_ring_brothel_hub
					}
				}
			}
		}
		planet_amenities_add = 2
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_tradition = tr_mercantile_trickle_up_economics }
		}
		trade_value_add = 1
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_trade_mult|
	}
	
	# Overlord
	triggered_planet_modifier = {
		potential = {
			planet = {
				NOT = { has_planet_flag = has_franchise_headquarters }
			}
		}
		trade_value_add = 4
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_trade_mult|
	}
	triggered_planet_modifier = {
		potential = {
			planet = {
				has_planet_flag = has_franchise_headquarters
			}
		}
		trade_value_add = 3
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_trade_mult|
	}
	overlord_resources = {
		category = planet_corporate_franchising_meddling
		produces = {
			trigger = {
				planet = {
					has_planet_flag = has_franchise_headquarters
				}
			}
			unity = 0.25
		}
	}
	
	weight = {
		weight = @clerk_job_weight
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_trade_mult|
		# factors
		modifier = {
			factor = 1.2
			has_job = lv_comfort_worker
		}
		modifier = {
			factor = 0.25
			is_enslaved = yes
			can_take_servant_job = no
		}
		modifier = {
			factor = 1.1
			exists = owner
			owner = { is_megacorp = yes }
		}
		modifier = {
			factor = 0.5
			exists = owner
			owner = { is_fallen_empire = yes }
		}
		
		# sex traits
		modifier = {
			add = 0.05
			lv_sex_weight_low = yes
		}
		modifier = {
			add = 0.1
			lv_sex_weight_med = yes
		}
		modifier = {
			add = 0.2
			lv_sex_weight_high = yes
		}
		modifier = {
			add = -0.2
			lv_sex_weight_neg = yes
		}
	}
}

lv_rod_temple_initiate = {			# Temple Initiate (event)
	category = worker
	condition_string = WORKER_JOB_TRIGGER
	building_icon = building_great_pyramid
	clothes_texture_index = 3

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_worker_job

	possible = {
		NOT = { pop_has_trait = trait_lv_equine_penis }
	}

	resources = {
		category = planet_lv_rod_temple_initiate
		
		# Organic
		upkeep = {
			trigger = {
				is_organic_species = yes
				is_lithoid = no
			}
			consumer_goods = 2
			food = 1
		}
		
		# Lithoids
		upkeep = {
			trigger = {
				is_lithoid = yes
			}
			consumer_goods = 2
			minerals = 1
		}
		
		# Any
		produces = {
			unity = 10
			society_research = 20
		}
	}

	# Any
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_add = 5
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	
	weight = {
		weight = @worker_job_weight
		mult = value:job_weights_multiple_resources_modifier|JOB|lv_rod_temple_initiate|RESOURCE1|unity|WEIGHT1|0.5|RESOURCE2|society_research|WEIGHT2|0.3|
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		# factors
		modifier = {
			factor = 0.5
			has_citizenship_rights = no  #willing apprentices first, please
			NOT = { has_trait = trait_mechanical }
		}
		modifier = {
			factor = 30
			NOT = { has_trait = trait_lv_equine_penis }
		}
		modifier = {
			factor = 0
			has_trait = trait_lv_equine_penis
		}
	}
}
