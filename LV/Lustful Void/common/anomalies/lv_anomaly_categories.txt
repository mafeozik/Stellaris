# The Mad Scientist's Tentacled Creation
lv_lewd_asteroids_1_category = {
	desc = "lv_lewd_asteroids_1_category_desc"
	picture = GFX_evt_mining_station
	level = 2
	
	max_once = yes
	
	spawn_chance = {		
		modifier = {
			add = 3
			is_asteroid = yes
		}
		modifier = {
			factor = 0
			from.owner = {
				is_machine_empire = yes
				owner.species = { lv_is_a_smexy_robutt = no }
			}
		}
		modifier = {
			factor = 0
			OR = {
				has_global_flag = lv_all_disabled
				has_global_flag = lv_bestiality_disabled
				has_global_flag = lv_rape_disabled
			}
		}
	}
	
	on_success = lv_lewd_asteroids_1.1
}
